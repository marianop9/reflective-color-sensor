# ws2812 driver ----------------------------------------------------------------
add_library(driver_ws2812 
    STATIC 
    ${CMAKE_CURRENT_LIST_DIR}/ws2812.c
)

# create output directory for generated PIO header:
file(MAKE_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/generated)

pico_generate_pio_header(driver_ws2812 
    # pio target:
    ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio
    # set output directory for generated PIO header:
    OUTPUT_DIR ${CMAKE_CURRENT_LIST_DIR}/generated
)

target_include_directories(driver_ws2812 
    PUBLIC
    ${CMAKE_CURRENT_LIST_DIR}/include
)

target_link_libraries(driver_ws2812 hardware_pio)

if (INCLUDE_WS2812_DEMO)
    add_executable(pio_ws2812_demo)

    # generate the header file into the source tree as it is included in the RP2040 datasheet
    # pico_generate_pio_header(pio_ws2812_demo ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio OUTPUT_DIR ${CMAKE_CURRENT_LIST_DIR}/generated)
    ## (Header was already generated by static library ABOVE)
    target_include_directories(pio_ws2812_demo 
        PUBLIC 
        ${CMAKE_CURRENT_LIST_DIR}/generated
    )

    target_sources(pio_ws2812_demo PRIVATE ws2812_demo.c)

endif()
