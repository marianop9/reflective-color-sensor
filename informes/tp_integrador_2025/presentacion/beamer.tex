\section{Circuito de Acondicionamiento}
\begin{frame}{Acondicionamiento de Señal}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\textwidth]{circ_acond.png}
    \end{figure}
    
    \begin{align*}
        V_o &= V_{+} + (V_{+} - V_{s}) \frac{R_f}{R_{LDR}} \\
        V_o &= V_{+} + (V_{+} - V_{s}) R_f G
    \end{align*}

    Mediante el CD4066, $R_f$ puede tomar distintos valores discretos para evitar la saturación ante el amplio rango del LDR (\qtyrange{1}{1000}{\kohm}). 
\end{frame}

\begin{frame}{Respuesta del Circuito}
    \begin{equation*}
        G = \frac{V_+ - V_o}{R_f * (V_s-V_+)}
    \end{equation*}

    \begin{figure}
        \includegraphics[width=0.7\textwidth]{circuit_response}
        \caption{Tensión de salida vs Resistencia del LDR para distintas $R_f$.}
    \end{figure}
\end{frame}

\section{Caracterización del LDR}
\begin{frame}{Modelo Matemático del LDR}
    La respuesta del LDR no es lineal:
    \begin{equation*}
        R = K E^{-\gamma}
    \end{equation*}
    \begin{itemize}
        \item $E$: \alert{Iluminancia}
        \item $Ka, \gamma$: Parámetros desconocidos.
    \end{itemize}
    \pause
    \vfill
    Expresado en terminos de conductancia:
    \begin{equation*}
        G = K_a E^{\gamma}
    \end{equation*}
\end{frame}

\begin{frame}{Iluminancia}
    \begin{block}{Definición}
        La iluminancia es la cantidad de flujo luminoso que incide sobre una superficie por unidad de área.
        
        \begin{itemize}
            \item Define qué tan iluminada está una superficie
            \item Medida en lux ($\qty{1}{lux} = \qty{1}{\lumen\per\meter^2}$)
            \item Disminuye con el cuadrado de la distancia a la fuente
            \item Se mide con un \alert{luxómetro}
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Modelo Matemático Aproximado del LDR}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            Es imposible determinar la \alert{iluminancia}, por lo que se utiliza una aproximación.
            
            \begin{equation*}
                E \approx c ,\quad c \in [0,255]
            \end{equation*}
            \begin{equation*}
                G = K_a c^{\gamma}
            \end{equation*}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{conductancia_intensidad_LSQ.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Cálculo de los Parametros del LDR}
Una vez recolectadas las muestras para cada color, se aplica la técnica de mínimos cuadrados (LSQ) para aproximar la respuesta.

La $i$-ésima muestra recolectada se expresa como:
\begin{equation*}
    G_i = K_a c_i^\gamma
\end{equation*}

\pause
\begin{equation}    
    \ln G_i = \ln K_a + \gamma \ln c
\end{equation}

\pause
Considerando $N$ muestras distintas, se arma un sistema de ecuaciones y se expresa en forma de matrices:

\begin{equation}
    \begin{bmatrix}
        \ln G_1    \\
        \ln G_2    \\
        \vdots \\
        \ln G_N
    \end{bmatrix}
    =
    \begin{bmatrix}
        1      & \ln c_1 \\
        1      & \ln c_2 \\
        \vdots & \vdots  \\
        1      & \ln c_N
    \end{bmatrix}
    \begin{bmatrix}
        \ln K_a \\
        \gamma
    \end{bmatrix}
\end{equation}

\end{frame}

\begin{frame}{Cálculo de los Parámetros del LDR}
Considerando $N$ muestras distintas, se arma un sistema de ecuaciones y se expresa en forma de matrices:

\begin{align*}
    \begin{bmatrix}
        \ln G_1    \\
        \ln G_2    \\
        \vdots \\
        \ln G_N
    \end{bmatrix}
    &=
    \begin{bmatrix}
        1      & \ln c_1 \\
        1      & \ln c_2 \\
        \vdots & \vdots  \\
        1      & \ln c_N
    \end{bmatrix}
    \begin{bmatrix}
        \ln K_a \\
        \gamma
    \end{bmatrix}
    \\
    Y &= A x
\end{align*}

Los valores de la matriz $x$ que mejor se ajustan a las muestras obtenidas se puede calcular como:
\begin{equation} \label{eq:LSQ_formula}
    x = {(A^T A)}^{-1} A^T Y
\end{equation}

\end{frame}

\section{Proceso de Medición}
\begin{frame}{Proceso de Medición}
    Una vez determinados los parámetros del LDR, se pueden tomar mediciones.

    Para cada color R-G-B:
    \begin{enumerate}
        \item Se coloca el dispositivo sobre la superficie a medir.
        \item Iluminación a máxima intensidad.
        \item Muestreo de tensión y cálculo de $G$.
    \end{enumerate}
    
    \[
        G \approx K_a c^\gamma
    \]

    \begin{itemize}
        \item $G$ se mide
        \item $K_a ,\, \gamma$ se conocen de la calibración
    \end{itemize}
\end{frame}

\begin{frame}{Arquitectura del Firmware}
    \begin{itemize}
        \item \texttt{usb\_task()} 
            \begin{itemize}
                \item Ejecuta del stack USB 
                \item Recibe/envía de mensajes desde/hacía el host
            \end{itemize}
        \item \texttt{worker\_task()} 
            \begin{itemize}
                \item Ejecuta de comandos
                \item Interactúa con periféricos a través de \textbf{drivers}
                \item Arma de mensajes de respuesta
            \end{itemize} 
        \item Librería \texttt{ascii-communication-protocol} (ACP) 
            \begin{itemize}
                \item Interpreta mensajes recibidos como comandos
                \item Codifica mensajes a enviar en el formato de respuesta
            \end{itemize}          
    \end{itemize}
